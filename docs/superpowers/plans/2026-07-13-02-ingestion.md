# Ingestion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Connect a tenant's GitHub repos to the app and turn their merged PRs and pushed commits into `ChangeItem` rows — the raw material later plans batch into AI-drafted updates.

**Architecture:** A GitHub App (not OAuth — this is a separate app-level integration, distinct from the login GitHub OAuth App from Plan 1) is installed per tenant via a "Connect GitHub" link that carries `state=<tenantId>`. GitHub redirects back to a setup route that lists the installation's accessible repos and creates `Repo` rows. From then on, GitHub sends webhooks (`pull_request`, `push`) to a single endpoint; after HMAC signature verification, each event is turned into one or more `ChangeItem` rows, scoped to the `Repo` (and therefore tenant) the event belongs to.

**Tech Stack:** `octokit` (GitHub App auth + REST client), `@octokit/webhooks-methods` (signature verification), Drizzle ORM (extending Plan 1's schema), Next.js Route Handlers.

## Global Constraints

- Every `ChangeItem` belongs to exactly one `Repo`, which belongs to exactly one `Tenant` — no cross-tenant data ever mixes in a single query result. (Design spec: multi-tenant model)
- `ChangeItem` is single-PR / single-commit by design — no batching or grouping at ingestion time. (Design spec: "Commit grouping" decision)
- Commit diffs are capped at ingestion time (~200 lines) before they're ever stored — large diffs must not flow further into the pipeline unbounded. (Design spec: "Generation Strategy" — truncation)
- `ChangeItem.updateId` is a plain nullable `uuid` column with **no foreign key constraint** in this plan — the `updates` table doesn't exist until Plan 3, which will add the constraint via its own migration. Do not add `.references()` on this column here.

```typescript
import { verify } from "@octokit/webhooks-methods";

export async function verifyGithubSignature(
  payload: string,
  signatureHeader: string | null
): Promise<boolean> {
  if (!signatureHeader) return false;

  const secret = process.env.GITHUB_WEBHOOK_SECRET;
  if (!secret) {
    throw new Error("GITHUB_WEBHOOK_SECRET is not set");
  }

  return verify(secret, payload, signatureHeader);
}
```

- [ ] **Step 5: Implement the GitHub App client and truncation helper**

Create `src/lib/github.ts`:

```typescript
import { App } from "octokit";

export const githubApp = new App({
  appId: process.env.GITHUB_APP_ID as string,
  privateKey: (process.env.GITHUB_APP_PRIVATE_KEY as string)?.replace(/\\n/g, "\n"),
});

export function truncateDiff(diff: string, maxLines = 200): string {
  const lines = diff.split("\n");
  if (lines.length <= maxLines) return diff;

  const kept = lines.slice(0, maxLines);
  const remaining = lines.length - maxLines;
  return [...kept, `… (truncated, ${remaining} more lines)`].join("\n");
}
```

Note the `.replace(/\\n/g, "\n")` on the private key — GitHub App private keys are PEM-formatted with real newlines, which most `.env` files can't hold literally; store it with escaped `\n` sequences in `.env.local` and this un-escapes it at load time.

```typescript
import { and, eq } from "drizzle-orm";
import { db as defaultDb } from "../db";
import { repos } from "../db/schema";

export async function syncReposForInstallation(
  tenantId: string,
  installationId: string,
  accessibleRepos: Array<{ full_name: string; default_branch: string }>,
  database: typeof defaultDb = defaultDb
): Promise<void> {
  for (const repo of accessibleRepos) {
    const existing = await database
      .select()
      .from(repos)
      .where(and(eq(repos.tenantId, tenantId), eq(repos.githubRepoFullName, repo.full_name)))
      .limit(1);

    if (existing.length > 0) {
      await database
        .update(repos)
        .set({ githubInstallationId: installationId, defaultBranch: repo.default_branch })
        .where(eq(repos.id, existing[0].id));
      continue;
    }

    await database.insert(repos).values({
      tenantId,
      githubRepoFullName: repo.full_name,
      githubInstallationId: installationId,
      defaultBranch: repo.default_branch,
      sourceTypes: ["pr"],
    });
  }
}
```

```typescript
import { NextRequest, NextResponse } from "next/server";
import { githubApp } from "@/lib/github";
import { syncReposForInstallation } from "@/lib/repo-sync";
import { requireSession } from "@/lib/session";

export async function GET(request: NextRequest) {
  const session = await requireSession();

  const installationId = request.nextUrl.searchParams.get("installation_id");
  const state = request.nextUrl.searchParams.get("state");

  if (!installationId || state !== session.user.tenantId) {
    return NextResponse.redirect(new URL("/?github_connect=error", request.url));
  }

  const installationOctokit = await githubApp.getInstallationOctokit(Number(installationId));
  const { data } = await installationOctokit.rest.apps.listReposAccessibleToInstallation();

  await syncReposForInstallation(
    session.user.tenantId,
    installationId,
    data.repositories.map((repo) => ({
      full_name: repo.full_name,
      default_branch: repo.default_branch ?? "main",
    }))
  );

  return NextResponse.redirect(new URL("/?github_connect=success", request.url));
}
```

```typescript
import { and, eq } from "drizzle-orm";
import { db as defaultDb } from "../db";
import { repos, changeItems } from "../db/schema";

export type MergedPullRequestInput = {
  installationId: string;
  repoFullName: string;
  prNumber: number;
  prTitle: string;
  prDescription: string;
  prUrl: string;
  mergedAt: Date;
};

export async function ingestMergedPullRequest(
  input: MergedPullRequestInput,
  database: typeof defaultDb = defaultDb
): Promise<void> {
  const [repo] = await database
    .select()
    .from(repos)
    .where(
      and(eq(repos.githubInstallationId, input.installationId), eq(repos.githubRepoFullName, input.repoFullName))
    )
    .limit(1);

  if (!repo || !repo.sourceTypes.includes("pr")) return;

  await database.insert(changeItems).values({
    tenantId: repo.tenantId,
    repoId: repo.id,
    sourceType: "pr",
    prNumber: input.prNumber,
    prTitle: input.prTitle,
    prDescription: input.prDescription,
    prUrl: input.prUrl,
    mergedAt: input.mergedAt,
  });
}
```

```typescript
import { NextRequest, NextResponse } from "next/server";
import { verifyGithubSignature } from "@/lib/github-webhook";
import { ingestMergedPullRequest } from "@/lib/ingest-pull-request";

export async function POST(request: NextRequest) {
  const rawBody = await request.text();
  const signature = request.headers.get("x-hub-signature-256");

  if (!(await verifyGithubSignature(rawBody, signature))) {
    return NextResponse.json({ error: "invalid signature" }, { status: 401 });
  }

  const event = request.headers.get("x-github-event");
  const payload = JSON.parse(rawBody);

  if (event === "pull_request" && payload.action === "closed" && payload.pull_request?.merged) {
    await ingestMergedPullRequest({
      installationId: String(payload.installation.id),
      repoFullName: payload.repository.full_name,
      prNumber: payload.pull_request.number,
      prTitle: payload.pull_request.title,
      prDescription: payload.pull_request.body ?? "",
      prUrl: payload.pull_request.html_url,
      mergedAt: new Date(payload.pull_request.merged_at),
    });
  }

  return NextResponse.json({ ok: true });
```

Create `src/lib/ingest-push.ts`:
```typescript
import { and, eq } from "drizzle-orm";
import { db as defaultDb } from "../db";
import { repos, changeItems } from "../db/schema";
import { truncateDiff } from "./github";

export type PushInput = {
  installationId: string;
  repoFullName: string;
  ref: string;
  commits: Array<{ id: string; message: string; url: string; timestamp: string }>;
};

export type GetCommitDiff = (owner: string, repo: string, sha: string) => Promise<string>;

export async function ingestPush(
  input: PushInput,
  getCommitDiff: GetCommitDiff,
  database: typeof defaultDb = defaultDb
): Promise<void> {
  const [repo] = await database
    .select()
    .from(repos)
    .where(
      and(eq(repos.githubInstallationId, input.installationId), eq(repos.githubRepoFullName, input.repoFullName))
    )
    .limit(1);

  if (!repo || !repo.sourceTypes.includes("commit")) return;
  if (input.ref !== `refs/heads/${repo.defaultBranch}`) return;

  const [owner, repoName] = input.repoFullName.split("/");

  for (const commit of input.commits) {
    const rawDiff = await getCommitDiff(owner, repoName, commit.id);

    await database.insert(changeItems).values({
      tenantId: repo.tenantId,
      repoId: repo.id,
      sourceType: "commit",
      commitSha: commit.id,
      commitMessage: commit.message,
      commitUrl: commit.url,
      committedAt: new Date(commit.timestamp),
      diff: truncateDiff(rawDiff),
    });
  }
}
```

Modify `src/app/api/webhooks/github/route.ts` — add the import and the `push` branch:
```typescript
import { NextRequest, NextResponse } from "next/server";
import { verifyGithubSignature } from "@/lib/github-webhook";
import { ingestMergedPullRequest } from "@/lib/ingest-pull-request";
import { ingestPush } from "@/lib/ingest-push";
import { githubApp } from "@/lib/github";

export async function POST(request: NextRequest) {
  const rawBody = await request.text();
  const signature = request.headers.get("x-hub-signature-256");

  if (!(await verifyGithubSignature(rawBody, signature))) {
    return NextResponse.json({ error: "invalid signature" }, { status: 401 });
  }

  const event = request.headers.get("x-github-event");
  const payload = JSON.parse(rawBody);

  if (event === "pull_request" && payload.action === "closed" && payload.pull_request?.merged) {
    await ingestMergedPullRequest({
      installationId: String(payload.installation.id),
      repoFullName: payload.repository.full_name,
      prNumber: payload.pull_request.number,
      prTitle: payload.pull_request.title,
      prDescription: payload.pull_request.body ?? "",
      prUrl: payload.pull_request.html_url,
      mergedAt: new Date(payload.pull_request.merged_at),
    });
  }

  if (event === "push") {
    const installationId = String(payload.installation.id);
    await ingestPush(
      {
        installationId,
        repoFullName: payload.repository.full_name,
        ref: payload.ref,
        commits: payload.commits.map((c: { id: string; message: string; url: string; timestamp: string }) => ({
          id: c.id,
          message: c.message,
          url: c.url,
          timestamp: c.timestamp,
        })),
      },
      async (owner, repoName, sha) => {
        const installationOctokit = await githubApp.getInstallationOctokit(Number(installationId));
        const { data: commit } = await installationOctokit.rest.repos.getCommit({ owner, repo: repoName, ref: sha });
        return (commit.files ?? []).map((f) => f.patch ?? "").join("\n");
      }
    );
  }

  return NextResponse.json({ ok: true });
}
```
